import React, { useRef, useState, useEffect } from "react";
import { Sidebar } from "./Sidebar";
import { Canvas } from "./Canvas";

const NODE_TYPES = {
  OBJECT_A: { id: "objectA", label: "Object A" },
  OBJECT_B: { id: "objectB", label: "Object B" },
};

export default function App() {
  const [objects, setObjects] = useState([]);
  const canvasRef = useRef(null);

  // World coordinate conversion using canvas bounding rect
  const screenToWorld = (mouseX, mouseY) => {
    if (!canvasRef.current) return { x: 0, y: 0 };
    const rect = canvasRef.current.getBoundingClientRect();
    return {
      x: (mouseX - rect.left) / 1,
      y: (mouseY - rect.top) / 1,
    };
  };

  // Add a new object at world coordinates
  const addObject = (type, worldX, worldY) => {
    const newObject = {
      id: `object-${Math.random().toString(36).substr(2, 9)}`,
      type,
      label: NODE_TYPES[type].label,
      x: worldX,
      y: worldY,
    };
    setObjects((prev) => [...prev, newObject]);
  };

  return (
    <div className="app">
      <Sidebar onAddObject={addObject} />
      <Canvas
        objects={objects}
        onAddObject={addObject}
        screenToWorld={screenToWorld}
      />
    </div>
  );
}
